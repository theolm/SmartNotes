cmake_minimum_required(VERSION 3.22)
project(smartnotes_whisper_probe LANGUAGES C CXX)

# Standalone qualification executable only. Never linked into the Flutter app.
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "The initial probe targets Android arm64-v8a only")
endif()
if(NOT WHISPER_SOURCE_DIR)
    message(FATAL_ERROR "Provide the verified, pinned WHISPER_SOURCE_DIR")
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH "armv8-a" CACHE STRING "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_CCACHE OFF CACHE BOOL "" FORCE)

if(GGML_VULKAN)
    find_package(Vulkan REQUIRED)
endif()
add_subdirectory("${WHISPER_SOURCE_DIR}" whisper)
if(GGML_VULKAN)
    target_include_directories(ggml-vulkan PRIVATE "${SPIRV_INCLUDE_DIR}")
endif()
add_executable(smartnotes-whisper-probe probe.cpp)
target_include_directories(smartnotes-whisper-probe PRIVATE "${WHISPER_SOURCE_DIR}/examples")
target_link_libraries(smartnotes-whisper-probe PRIVATE whisper)
if(GGML_VULKAN)
    target_link_libraries(smartnotes-whisper-probe PRIVATE Vulkan::Vulkan)
endif()
target_compile_definitions(smartnotes-whisper-probe PRIVATE
    SMARTNOTES_PROBE_VULKAN=$<BOOL:${GGML_VULKAN}>)
